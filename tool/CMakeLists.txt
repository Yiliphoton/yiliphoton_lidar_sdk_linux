cmake_minimum_required(VERSION 3.5)
project(ylp_driver_tools)

message("-- Building tools")
include_directories(${DRIVER_INCLUDE_DIRS})

if(COMPILE_TOOL_VIEWER OR COMPILE_TOOLS)
  find_package(PCL COMPONENTS common visualization io QUIET)
  if(PCL_FOUND)
    add_definitions(${PCL_DEFINITIONS})
    include_directories(${PCL_INCLUDE_DIRS})
    link_directories(${PCL_LIBRARY_DIRS})
    add_executable(simple_viewer simple_viewer.cpp)
    target_link_libraries(simple_viewer ${EXTERNAL_LIBS} ${PCL_LIBRARIES})
    install(TARGETS simple_viewer RUNTIME DESTINATION /usr/bin)
  else()
    message(WARNING "PCL not found — simple_viewer will not be built")
  endif()
endif()
